% Fig. 1: VeriLoop architecture (TikZ, included by main.tex)
\begin{tikzpicture}[
  font=\sffamily\scriptsize,
  stage/.style={draw=black!70, rounded corners=2pt, fill=white, align=center, text width=2.55cm,
                minimum height=1.55cm, inner sep=3pt},
  where/.style={rounded corners=2pt, align=center, text width=2.55cm, minimum height=0.5cm,
                font=\sffamily\scriptsize\bfseries, text=black!75},
  arr/.style={-{Stealth[length=5pt]}, thick, black!75},
  fb/.style={-{Stealth[length=5pt]}, thick, red!65!black, dashed},
  lab/.style={font=\sffamily\tiny, text=red!55!black, fill=white, inner sep=1pt}
]
\def\dx{3.25}
\foreach \i/\name/\col/\loc in {0/sense/blue!10/Wearable device,
                                1/interp/green!12/Edge gateway (home),
                                2/decide/green!12/Edge gateway (home),
                                3/exec/orange!14/{Patient \& care team},
                                4/valid/red!8/Edge + fog}
  \node[where, fill=\col] (w\name) at (\i*\dx, 1.25) {\loc};

\node[stage] (sense)  at (0*\dx,0) {\textbf{1 Sense}\\[1pt]HR, RR, SpO$_2$, SBP, Temp;\\adaptive 5/15-min duty cycle};
\node[stage] (interp) at (1*\dx,0) {\textbf{2 Interpret}\\[1pt]signal-quality \& fault diagnosis; personal baseline; risk model};
\node[stage] (decide) at (2*\dx,0) {\textbf{3 Decide}\\[1pt]tiered service contract, NEWS2 red-flag guardrails};
\node[stage] (exec)   at (3*\dx,0) {\textbf{4 Execute}\\[1pt]re-measure, device check, teleconsult, escalation};
\node[stage, fill=red!3] (valid) at (4*\dx,0) {\textbf{5 Validate}\\[1pt]confirm data, verify outcome, adapt bounded rules};

\draw[arr] (sense) -- (interp);
\draw[arr] (interp) -- (decide);
\draw[arr] (decide) -- (exec);
\draw[arr] (exec) -- (valid);

% feedback loops below the stages
\draw[fb] ([xshift=-6mm]valid.south) -- ++(0,-0.45) -| node[lab, pos=0.25]{outcome not verified $\rightarrow$ escalate} (decide.south);
\draw[fb] (valid.south) -- ++(0,-0.95) -| node[lab, pos=0.22]{false alarm $\rightarrow$ raise personal threshold, refresh baseline (bounded)} (interp.south);
\draw[fb] ([xshift=6mm]valid.south) -- ++(0,-1.45) -| node[lab, pos=0.15]{data not confirmed $\rightarrow$ device check $\rightarrow$ probe re-seated} (sense.south);

% ledger
\node[draw=black!60, fill=gray!10, rounded corners=2pt, minimum width=4*\dx cm+2.65cm, minimum height=0.5cm,
      anchor=north, align=center] (led) at (2*\dx,-2.45)
      {\textbf{Provenance ledger}: every loop iteration appended to a SHA-256 hash chain at the edge; hourly anchor hashes sent to the cloud; FHIR \texttt{Observation}/\texttt{ServiceRequest}/\texttt{Task}/\texttt{Provenance} export};
\end{tikzpicture}
